5) RBAC/Scopes: enforce on API, not only UI

Prompt (policy file):

// rbac/policies.ts
export const Policies = {
  "payroll:run:read": ["admin","payroll_mgr","auditor"],
  "payroll:run:execute": ["admin","payroll_mgr"],
  "filings:submit": ["admin","accountant"],
};
// middleware
export const requireScope = (scope:string) => (req,res,next)=>{
  const roles = req.user?.roles || [];
  return Policies[scope]?.some(r=>roles.includes(r)) ? next() : res.sendStatus(403);
};


Usage:
