🔧 One migration to fix all (copy-paste)

BEGIN;

-- Disaster Recovery additions
ALTER TABLE IF EXISTS public.dr_slas
  ADD COLUMN IF NOT EXISTS last_tested_at timestamptz;

ALTER TABLE IF EXISTS public.restore_tests
  ADD COLUMN IF NOT EXISTS started_at timestamptz;

ALTER TABLE IF EXISTS public.worm_objects
  ADD COLUMN IF NOT EXISTS compliance_policy jsonb;

-- Backfill sensible defaults where null to satisfy code paths
UPDATE public.worm_objects
SET compliance_policy = '{}'::jsonb
WHERE compliance_policy IS NULL;

-- On-call / Policies
ALTER TABLE IF EXISTS public.escalation_policies
  ADD COLUMN IF NOT EXISTS escalation_steps jsonb;

UPDATE public.escalation_policies
SET escalation_steps = '[]'::jsonb
WHERE escalation_steps IS NULL;

-- Automated runbooks / trigger state
ALTER TABLE IF EXISTS public.runbook_triggers
  ADD COLUMN IF NOT EXISTS system_states jsonb;

UPDATE public.runbook_triggers
SET system_states = '{}'::jsonb
WHERE system_states IS NULL;

-- Helpful GIN indexes for JSON lookups (safe if unused)
CREATE INDEX IF NOT EXISTS idx_escalation_policies_steps_gin
  ON public.escalation_policies USING gin (coalesce(escalation_steps, '[]'::jsonb));

CREATE INDEX IF NOT EXISTS idx_runbook_triggers_states_gin
  ON public.runbook_triggers USING gin (coalesce(system_states, '{}'::jsonb));

COMMIT;
