Data Model (key tables)
-- Partner firm & membership
create table partner_firm (
  id uuid primary key,
  name text not null,
  vat text, address text, status text default 'active',
  created_at timestamptz default now()
);
create table partner_member (
  id uuid primary key,
  partner_id uuid references partner_firm(id),
  user_id uuid not null,
  role text check (role in ('partner_admin','partner_staff','partner_reviewer','partner_viewer')),
  created_at timestamptz default now()
);

-- Link partner ↔ client tenant + scopes
create table partner_client_link (
  id uuid primary key,
  partner_id uuid references partner_firm(id),
  tenant_id uuid not null,
  status text check (status in ('pending','active','revoked')) default 'pending',
  scopes jsonb not null,     -- e.g. ["filings:prepare","filings:submit","runs:view","audit:download"]
  maker_checker_mode text check (maker_checker_mode in ('client_checker','partner_checker','dual')) default 'client_checker',
  created_at timestamptz default now(),
  activated_at timestamptz
);

-- Approvals (maker-checker)
create table approval_request (
  id uuid primary key,
  tenant_id uuid not null,
  created_by uuid not null,              -- partner_member.user_id or client user
  created_by_type text check (created_by_type in ('partner','client')),
  action text not null,                  -- e.g. "apd.submit", "fmy.submit", "payroll.run.finalize"
  payload jsonb not null,
  status text check (status in ('pending','approved','rejected','expired')) default 'pending',
  approver_role text,                    -- 'client_checker' or 'partner_reviewer'
  approved_by uuid,
  approved_at timestamptz,
  reason text,
  created_at timestamptz default now()
);

-- OBO audit
create table audit_event (
  id uuid primary key,
  tenant_id uuid not null,
  actor_user_id uuid not null,
  actor_type text check (actor_type in ('partner','client')) not null,
  partner_id uuid,
  action text not null,                  -- e.g. "ERGANI.SUBMIT", "APD.SUBMIT"
  object_type text, object_id text,
  metadata jsonb,
  at timestamptz default now(),
  hash bytea
);
