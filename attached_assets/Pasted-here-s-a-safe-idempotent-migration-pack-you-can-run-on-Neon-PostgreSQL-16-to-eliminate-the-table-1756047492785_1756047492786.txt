-- === Optional services bootstrap (idempotent) ===
-- Target: Neon / PostgreSQL 16

BEGIN;

-- Some objects below use gen_random_uuid()
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- 1) LOGGING: ensure priority column exists on log_retention_policies
ALTER TABLE IF EXISTS public.log_retention_policies
  ADD COLUMN IF NOT EXISTS priority integer;

-- backfill + constraint safely
UPDATE public.log_retention_policies
SET priority = 0
WHERE priority IS NULL;

ALTER TABLE public.log_retention_policies
  ALTER COLUMN priority SET NOT NULL,
  ALTER COLUMN priority SET DEFAULT 0;

-- 2) RUNBOOKS: ensure dependencies jsonb on runbooks
ALTER TABLE IF EXISTS public.runbooks
  ADD COLUMN IF NOT EXISTS dependencies jsonb;

-- backfill + default
UPDATE public.runbooks
SET dependencies = '[]'::jsonb
WHERE dependencies IS NULL;

ALTER TABLE public.runbooks
  ALTER COLUMN dependencies SET DEFAULT '[]'::jsonb;

-- Optional: enforce NOT NULL if your service expects it
-- ALTER TABLE public.runbooks ALTER COLUMN dependencies SET NOT NULL;

-- helpful GIN index if you query dependencies content
CREATE INDEX IF NOT EXISTS idx_runbooks_dependencies_gin
  ON public.runbooks USING gin ((coalesce(dependencies, '[]'::jsonb)));

-- 3) ON-CALL: create oncall_rotas table if missing
CREATE TABLE IF NOT EXISTS public.oncall_rotas (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  team_id uuid NOT NULL,
  name text NOT NULL,
  timezone text NOT NULL DEFAULT 'Europe/Athens',
  starts_at timestamptz NOT NULL,
  rotation_length_days integer NOT NULL DEFAULT 7,
  escalation_policy jsonb NOT NULL DEFAULT '[]'::jsonb,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_oncall_rotas_team_id
  ON public.oncall_rotas(team_id);

CREATE INDEX IF NOT EXISTS idx_oncall_rotas_starts_at
  ON public.oncall_rotas(starts_at);

-- 4) AUTOMATED RUNBOOKS: create table if your service expects it
CREATE TABLE IF NOT EXISTS public.automated_runbooks (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text NOT NULL,
  description text,
  active boolean NOT NULL DEFAULT true,
  trigger_type text NOT NULL DEFAULT 'manual', -- manual|schedule|event
  schedule_cron text,                          -- if trigger_type='schedule'
  dependencies jsonb NOT NULL DEFAULT '[]'::jsonb,
  owner_user_id uuid,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_automated_runbooks_active
  ON public.automated_runbooks(active);

CREATE INDEX IF NOT EXISTS idx_automated_runbooks_owner
  ON public.automated_runbooks(owner_user_id);

COMMIT;
