// Bank profile & cut-offs
type BankProfile = {
  id: "alpha" | "piraeus" | "eurobank" | "nbg";
  painVersion: "pain.001.001.03" | "pain.001.001.09";
  supportsInstant: boolean;                  // SCT Inst capability
  sctInstAmountLimit?: number;               // bank/rail limit (e.g., 100000)
  cutoffs: {                                 // local times
    sct: { weekdays: string; time: "HH:mm" },
    sctInst: { alwaysOn: boolean }           // usually true
  };
};

// Batch (one payroll run may create multiple batches per bank/profile)
type PaymentBatch = {
  batchId: string;
  tenantId: string;
  entityId: string;
  bankProfileId: BankProfile["id"];
  runId: string;                             // payroll run
  method: "SCT" | "SCT_INST";
  status: "prepared" | "submitted" | "accepted" | "partially_settled" | "settled" | "reconciled" | "failed";
  totals: { count: number; amount: number; accepted?: number; rejected?: number; settled?: number };
  submittedAt?: string;                      // ISO
  fileRefs: { pain001?: string; bankReceipt?: string }[];
  supersedes?: string[];                     // previous batchIds superseded
  metadata?: Record<string, string>;
};

// Line-level
type PaymentInstruction = {
  lineId: string;
  batchId: string;
  employeeId: string;
  endToEndId: string;                        // unique per instruction
  amount: number;
  currency: "EUR";
  creditorName: string;
  creditorIban: string;
  method: "SCT" | "SCT_INST";
  status: "prepared" | "submitted" | "accepted" | "settled" | "rejected" | "superseded" | "cancelled";
  reasonCode?: string;                       // from pain.002 (e.g., AM04, AC04, FF01)
  bankRefs?: { uetr?: string; // if available for inst
              bankTxId?: string };
  originalLineId?: string;                   // set for re-issues
  reconciliation: {
    pain002?: { fileId: string; ts: string; status: string; reasonCode?: string };
    camt?: { fileId: string; ts: string; amount: number; bookingDate: string };
  };
};
