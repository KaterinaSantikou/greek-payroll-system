3) Make logging/audit non-fatal (so missing fields never crash again)
3a) CentralLogAggregationService: swallow DB write errors
-  async log(entry: LogEntry): Promise<void> {
-    await this.client.queryObject(`INSERT INTO log_entries (...) VALUES (...)`, [ ... ]);
-  }
+  async log(entry: LogEntry): Promise<void> {
+    try {
+      await this.client.queryObject(`INSERT INTO log_entries (...) VALUES (...)`, [ ... ]);
+    } catch (e:any) {
+      console.warn('[Logging][soft-fail]', e.code || e.message, { path: entry.path, service: entry.service });
+      // Never throw – console is the fallback sink
+    }
+  }

3b) AuditService.logEvent: soft-fail unless STRICT_AUDIT=true
-  static async logEvent(ev: AuditEvent) {
-    await db.queryObject(`INSERT INTO audit_events (...) VALUES (...)`);
-  }
+  static async logEvent(ev: AuditEvent) {
+    try {
+      await db.queryObject(`INSERT INTO audit_events (...) VALUES (...)`);
+    } catch (e:any) {
+      const strict = process.env.STRICT_AUDIT === 'true';
+      const msg = e.code || e.message;
+      if (strict) throw new Error(`Audit hard-fail: ${msg}`);
+      console.warn('[Audit][soft-fail]', msg, { action: ev.action, entity: ev.entity_type });
+    }
+  }

3c) GDPRComplianceInitializer: don’t abort boot on audit soft-fail
- await AuditService.initialize(); // throws on failure
+ try {
+   await AuditService.initialize();
+ } catch (e:any) {
+   const strict = process.env.STRICT_GDPR_INIT === 'true';
+   if (strict) throw e;
+   console.warn('[GDPR][soft-fail] Audit init', e.message || e);
+ }


These three prevent a single missing column from crashing the whole app again.
