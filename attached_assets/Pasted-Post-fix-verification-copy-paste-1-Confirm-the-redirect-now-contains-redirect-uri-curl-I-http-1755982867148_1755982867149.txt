LOC=$(curl -sI "https://<YOUR-REPLIT-DOMAIN>/api/login" -H "X-Forwarded-Host: <YOUR-REPLIT-DOMAIN>" | tr -d '\r' | awk '/^Location:/ {print $2}')

import sys, urllib.parse as u
loc=sys.stdin.read().strip()

params=dict(u.parse_qsl(q))
print("redirect_uri (decoded):", params.get("redirect_uri"))
PY <<< "$LOC"
# Expect: https://<YOUR-REPLIT-DOMAIN>/oauth2callback
