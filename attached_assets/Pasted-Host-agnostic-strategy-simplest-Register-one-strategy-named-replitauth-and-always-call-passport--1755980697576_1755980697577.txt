Some Passport OIDC/OAuth strategy implementations read callbackURL only at construction. If yours does, add a tiny wrapper to override callbackURL just before redirect:

// middleware before passport.authenticate('replitauth', ...)
app.use("/api/login", (req, _res, next) => {
  const host = (req.headers["x-forwarded-host"] as string) || req.headers.host || req.hostname;
  const cb = `${req.protocol}://${host}/api/callback`;
  // @ts-ignore accessing internal strategy object:
  const strat: any = (passport as any)._strategies["replitauth"];
  if (strat && strat._oauth2) strat._callbackURL = cb;
  if (strat && strat._options) strat._options.callbackURL = cb;
  next();
});
